QUESTION 39

Write a program on & quot raw & quot AJAX implementation using the XMLHttpRequest
$ cat solution
XMLHttpRequest object to fetch and display data from an XML file.  
data.xml 
<?xml version="1.0" encoding="UTF-8"?> 
<books> 
  <book> 
    <title>The Alchemist</title> 

 
    <author>Paulo Coelho</author> 
  </book> 
  <book> 
    <title>Harry Potter</title> 
    <author>J.K. Rowling</author> 
  </book> 
</books> 
 
index.html 
<!DOCTYPE html> 
<html> 
<head> 
  <title>AJAX XML Data Fetch</title> 
  <style> 
    body { font-family: Arial; padding: 20px; } 
    .book { margin-bottom: 10px; } 
  </style> 
</head> 
<body> 
  <h2>Books from XML</h2> 
  <button onclick="loadXMLData()">Load Books</button> 
  <div id="output"></div> 
 
  <script> 
    function loadXMLData() { 
      // Create an XMLHttpRequest object 
      var xhr = new XMLHttpRequest(); 
 
      // Define the request 
      xhr.open("GET", "data.xml", true); 
 
      // When the request is complete 
      xhr.onload = function() { 
        if (xhr.status === 200) { 
          var xml = xhr.responseXML; 
          var books = xml.getElementsByTagName("book"); 
          var output = ""; 
 
          for (var i = 0; i < books.length; i++) { 
            var title = books[i].getElementsByTagName("title")[0].textContent; 
            var author = books[i].getElementsByTagName("author")[0].textContent; 
 
            output += "<div class='book'><strong>" + title + "</strong><br>by " + author + 
"</div>"; 
          } 
 
        document.getElementById("output").innerHTML = output; 
        } else { 

 
          document.getElementById("output").innerHTML = "Error fetching data."; 
        } 
      };  
 
      // Send the request 
      xhr.send(); 
    } 
  </script> 
</body> 
</html>
