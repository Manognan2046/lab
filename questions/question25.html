QUESTION 25

Write a function that checks if a number is an integer and returns a Boolean value
$ cat solution
indicating whether it is true or false. 
<!DOCTYPE html> 
<html> 
<head> 
  <title>Check if Number is Integer</title> 
  <script> 
    function isInteger(num) { 
      return Number.isInteger(num); 
       
      // Alternatively, manual check: 
      // return typeof num === 'number' && num % 1 === 0; 

 
    } 
 
    function checkNumber() { 
      var input = document.getElementById("numInput").value; 
      var number = Number(input); // convert input to number 
      var result = isInteger(number); 
      document.getElementById("result").innerHTML =  
        "Is " + input + " an integer? -> " + result; 
    } 
  </script> 
</head> 
<body> 
  <h2>Check if a Number is Integer</h2> 
  <input type="text" id="numInput" placeholder="Enter a number"> 
  <button onclick="checkNumber()">Check</button> 
 
  <p id="result"></p> 
</body> 
</html>
