# program.py 
def process_instruction(line): 
    parts = line.strip().split() 
    if not parts: 

 
        return "Invalid instruction" 
    command = parts[0].upper() 
    try: 
        if command == "ADD": 
            return f"Result: {int(parts[1]) + int(parts[2])}" 
        elif command == "SUBTRACT": 
            return f"Result: {int(parts[1]) - int(parts[2])}" 
        elif command == "MULTIPLY": 
            return f"Result: {int(parts[1]) * int(parts[2])}" 
        elif command == "DIVIDE": 
            return f"Result: {int(parts[1]) / int(parts[2])}" 
        elif command == "REVERSE": 
            return f"Result: {parts[1][::-1]}" 
        elif command == "UPPERCASE": 
            return f"Result: {parts[1].upper()}" 
        else: 
            return "Unknown command" 
    except (IndexError, ValueError, ZeroDivisionError) as e: 
        return f"Error: {str(e)}" 
def main(): 
    input_file = "input.txt" 
    output_file = "output.txt" 
    with open(input_file, "r") as infile, open(output_file, "w") as outfile: 
        for line in infile: 
            result = process_instruction(line) 
            outfile.write(result + "\n") 
    print(f"Instructions processed. Results saved to '{output_file}'.") 
if __name__ == "__main__": 

 
    main()
