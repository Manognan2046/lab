QUESTION 17

Create a student Registration form for Job Application and validate the form fields using
$ cat solution
using JavaScript.  
<html> 
  <body> 
    <h2>Student Job Application Form</h2> 
    <form onsubmit="return validateForm()"> 
      Name: <input type="text" id="name"><br><br> 
      Email: <input type="text" id="email"><br><br> 

 
      Phone: <input type="text" id="phone"><br><br> 
     Gender: 
      <select id="gender"> 
        <option value="">Select</option> 
        <option>Male</option> 
        <option>Female</option> 
        <option>Other</option> 
      </select><br><br> 
     Date of Birth: <input type="date" id="dob"><br><br> 
     Qualification: 
      <select id="qualification"> 
        <option value="">Select</option> 
        <option>B.Tech</option> 
        <option>B.Sc</option> 
        <option>MCA</option> 
      </select><br><br> 
     Upload Resume: <input type="file" id="resume"><br><br> 
     Password: <input type="password" id="password"><br><br> 
      <p id="error" style="color:red;"></p> 
           <input type="submit" value="Register"> 
    </form> 
   <script> 
      function validateForm() { 
        var name = document.getElementById("name").value; 
        var email = document.getElementById("email").value; 
        var phone = document.getElementById("phone").value; 
        var gender = document.getElementById("gender").value; 
        var dob = document.getElementById("dob").value; 

 
        var qualification = document.getElementById("qualification").value; 
        var resume = document.getElementById("resume").value; 
        var password = document.getElementById("password").value; 
        var error = document.getElementById("error"); 
 
        if ( 
          name === "" || 
          email === "" || 
          phone === "" || 
          gender === "" || 
          dob === "" || 
          qualification === "" || 
          resume === "" || 
          password === "" 
        ) { 
          error.innerHTML = "Please fill all fields."; 
          return false; 
        } 
 
      if (!email.includes("@") || !email.includes(".")) { 
          error.innerHTML = "Invalid email."; 
          return false; 
        } 
       if (phone.length !== 10) { 
          error.innerHTML = "Phone must be 10 digits."; 
          return false; 
        } 
         if (password.length < 6) { 

 
          error.innerHTML = "Password must be at least 6 characters."; 
          return false; 
        } 
         alert("Form submitted successfully!"); 
        return true; 
      } 
    </script> 
  </body> 
</html>
