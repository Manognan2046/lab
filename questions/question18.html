Question 18

Develop and demonstrate a HTML file that includes Java script for the following problems

Code / Solution from the Record

problems input: A number n obtained using prompt Output: The first n Fibonacci 
numbers 
<html> 
  <body> 
    <h3>Fibonacci Numbers</h3> 
    <script> 
      var n = parseInt(prompt("Enter how many Fibonacci numbers to print:")); 

 
      if ( n <= 0) { 
        document.write("Please enter a valid positive number."); 
      } else { 
        var a = 0, b = 1, c; 
        var result = ""; 
      for (var i = 0; i < n; i++) { 
          result += a + " "; 
          c = a + b; 
          a = b; 
          b = c; 
        } 
       document.write("First " + n + " Fibonacci numbers:<br>" + result); 
      } 
    </script> 
  </body> 
</html>
