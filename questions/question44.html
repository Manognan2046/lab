Question 44

Develop a simple Java Servlet application

Code / Solution from the Record

SimpleServletApp/ 
├── WEB-INF/ 
│   ├── web.xml 
│   └── classes/ 
│       └── HelloServlet.class 
 
I.
HelloServlet.java 
import java.io.*; 
import javax.servlet.*; 
import javax.servlet.http.*; 
 

 
public class HelloServlet extends HttpServlet { 
    public void doGet(HttpServletRequest request, HttpServletResponse response) 
            throws ServletException, IOException { 
 
        response.setContentType("text/html"); 
        PrintWriter out = response.getWriter(); 
        out.println("<h2>Hello from Servlet!</h2>"); 
    } 
} 
 
II.
web.xml 
<web-app xmlns="http://java.sun.com/xml/ns/javaee" 
         version="3.0"> 
    <servlet> 
        <servlet-name>HelloServlet</servlet-name> 
        <servlet-class>HelloServlet</servlet-class> 
    </servlet> 
    <servlet-mapping> 
        <servlet-name>HelloServlet</servlet-name> 
        <url-pattern>/hello</url-pattern> 
    </servlet-mapping> 
</web-app> 
 
III.
Compile  
javac -classpath "path_to_tomcat/lib/servlet-api.jar" HelloServlet.java 
 
IV.
Run in tomcat 
http://localhost:8080/SimpleServletApp/hello
