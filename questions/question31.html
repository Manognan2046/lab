Question 31

Write a program to encrypt the given number to display the encrypted data using Java

Code / Solution from the Record

Java Script 
 <!DOCTYPE html> 
<html> 
<head> 
  <title>Encrypt Number</title> 

 
</head> 
<body style="font-family: Arial; text-align: center; padding-top: 50px;"> 
<h2>Encrypt a 4-digit Number</h2> 
  <input type="text" id="inputNumber" maxlength="4" placeholder="Enter 4-digit number"> 
  <button onclick="encrypt()">Encrypt</button> 
  <h3 id="result"></h3> 
  <script> 
    function encrypt() { 
      let num = document.getElementById("inputNumber").value; 
      if (num.length !== 4 || isNaN(num)) { 
        alert("Please enter a valid 4-digit number"); 
        return; 
      } 
      let digits = num.split('').map(Number); 
      for (let i = 0; i < 4; i++) { 
        digits[i] = (digits[i] + 7) % 10; 
      } 
      // Step 2: Swap 1st and 3rd, 2nd and 4th 
      [digits[0], digits[2]] = [digits[2], digits[0]]; 
      [digits[1], digits[3]] = [digits[3], digits[1]]; 
      let encrypted = digits.join(''); 
      document.getElementById("result").innerText = "Encrypted Number: " + encrypted; 
    } 
  </script> 
</body> 
</html>
