QUESTION 24

Write a program that uses the String object to reverse a given string.

s_reverse.html 
<!DOCTYPE html> 
<html> 
      <head> 
  
  <title>String Reverser</title> 
      </head> 
      <body style="text-align: center; margin-top: 50px;"> 
  
  <h2>String Reverser</h2> 
  
  <label for="inputString">Enter a string:</label> 
  
  <input type="text" id="inputString" /> 

 
              <button onclick="reverseString()">Reverse</button> 
  
  <p id="result"></p> 
  
  <script> 
   
      function reverseString() { 
      
         const str = document.getElementById("inputString").value; 
                     const reversed = str.split('').reverse().join(''); 
      
         document.getElementById("result").innerText = "Reversed string: " + reversed;    
} 
  
  </script> 
      </body> 
</html>
