Question 12

Write an HTML program that contains a selection box with a list of 5 countries. When the

Code / Solution from the Record

the user selects a country, its capital should be printed next to the list. Add CSS to 
customize the properties of the font of the capital (color, bold and font size). 
 
<!DOCTYPE html> 
<html lang="en"> 
<head> 
    <title>Country and Capital</title> 
    <style> 
        body { 
            font-family: Arial, sans-serif; 
            margin: 20px; 
        } 
        select { 
            padding: 10px; 
            font-size: 16px; 
            margin-right: 10px; 
        } 
        .capital { 
            font-weight: bold; 
            font-size: 18px; 
            color: #007BFF; /* Blue color */ 
        } 
    </style> 
</head> 
<body> 
    <h1>Country and Capital</h1> 
    <label for="countrySelect">Select a country:</label> 
    <select id="countrySelect" onchange="showCapital()"> 
        <option value="" disabled selected>Select a country</option> 
        <option value="India">India</option> 
        <option value="USA">USA</option> 
        <option value="France">France</option> 
        <option value="Japan">Japan</option> 
        <option value="Australia">Australia</option> 
    </select> 
    <span class="capital" id="capitalDisplay"></span> 
    <script> 
        function showCapital() { 
            const countryToCapital = { 
                "India": "New Delhi", 
                "USA": "Washington, D.C.", 
                "France": "Paris", 
                "Japan": "Tokyo", 
                "Australia": "Canberra" 
            }; 
            const selectedCountry = document.getElementById("countrySelect").value; 

 
            const capitalDisplay = document.getElementById("capitalDisplay"); 
  
            capitalDisplay.textContent = countryToCapital[selectedCountry] || ""; 
        } 
    </script> 
</body> 
</html>
