import org.w3c.dom.*;  
import javax.xml.parsers.*;  
import java.io.File;  
import java.util.Scanner;  
public class UserDetails {  
public static void main(String[] args) {  
try {  
File xmlFile = new File("users.xml");  
DocumentBuilderFactory factory =  
DocumentBuilderFactory.newInstance();  
DocumentBuilder builder = factory.newDocumentBuilder();  
Document document = builder.parse(xmlFile);  
document.getDocumentElement().normalize();  
NodeList userList = document.getElementsByTagName("user");  
Scanner scanner = new Scanner(System.in);  
System.out.print("Enter User ID to fetch details: ");  
String userId = scanner.nextLine();  
boolean userFound = false;  
for (int i = 0; i < userList.getLength(); i++) {  
Node userNode = userList.item(i);  
if (userNode.getNodeType() == Node.ELEMENT_NODE) {  
Element userElement = (Element) userNode;  
String id =  
userElement.getElementsByTagName("id").item(0).getTextContent();  
if (id.equals(userId)) {  
System.out.println("User Details:");  
System.out.println("ID: " + id);  
System.out.println("Name: " +  
userElement.getElementsByTagName("name").item(0).getTextContent());  
System.out.println("Email: " +  
userElement.getElementsByTagName("email").item(0).getTextContent());  
System.out.println("Phone: " +  
userElement.getElementsByTagName("phone").item(0).getTextContent());  
System.out.println("Address: " +  
userElement.getElementsByTagName("address").item(0).getTextContent());  
userFound = true;  
break;  
}  
}  
}
