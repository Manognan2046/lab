adds a new element to the end, and logs the modified array. in simple and easier way 
<html>

    <h3>Array operation</h3> 
<script> 
      var input = prompt("Enter numbers separated by commas:"); 

 
      var arr = input.split(",");  
      document.write("Original Array: " + input + "<br><br>"); 
      var newElement = prompt("Enter a number to add at the end:"); 
      arr.shift();            
      arr.push(newElement);  
      document.write("Modified Array: " + arr); 
    </script>
