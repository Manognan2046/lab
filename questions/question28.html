QUESTION 28

Write a Java script with string processing and regular expressions
$ cat solution
<!DOCTYPE html> 
<html> 
<head> 
  <title>String Processing and RegEx Demo</title> 
  <script> 
    function processString() { 
      var input = prompt("Enter a sentence:"); 
 
      // 1. Check if string contains only alphabets and spaces 
      var alphaRegex = /^[A-Za-z\s]+$/; 
      var isAlphabetic = alphaRegex.test(input); 
 
      // 2. Replace multiple spaces with a single space 
      var cleaned = input.replace(/\s+/g, ' ').trim(); 
 
      // 3. Extract all words into an array 
      var words = cleaned.split(" "); 
 
      // Prepare output message 
      var message = "Original Input: " + input + "\n"; 
      message += "Contains only alphabets and spaces? " + isAlphabetic + "\n"; 
      message += "Cleaned String (single spaces): " + cleaned + "\n"; 
      message += "Words Extracted (" + words.length + "): " + words.join(", "); 
 
      // Display result using alert 
      alert(message); 
    } 
  </script> 

 
</head> 
 
<body onload="processString()"> 
  <h2>JavaScript String Processing with RegEx</h2> 
  <p>Check the alert box for output when page loads.</p> 
</body> 
</html>
