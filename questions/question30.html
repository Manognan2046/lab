        function convertToWords() { 
            const numberInput = document.getElementById("number").value.trim(); 
            const outputDiv = document.getElementById("output"); 
            outputDiv.textContent = ""; 
            if (!/^\d+$/.test(numberInput)) { 
                outputDiv.textContent = "Invalid input. Please enter a valid number (0-999)."; 
                return; 
            } 
            const number = parseInt(numberInput, 10); 
            if (number < 0 || number > 999) { 
                outputDiv.textContent = "Number out of range. Please enter a number between 0 
and 999."; 
                return; 
            } 
            const singleDigits = ["", "One", "Two", "Three", "Four", "Five", "Six", "Seven", "Eight", 
"Nine"]; 
            const teens = ["Ten", "Eleven", "Twelve", "Thirteen", "Fourteen", "Fifteen", "Sixteen", 
"Seventeen", "Eighteen", "Nineteen"]; 
            const tens = ["", "", "Twenty", "Thirty", "Forty", "Fifty", "Sixty", "Seventy", "Eighty", 
"Ninety"]; 
            let words = ""; 
            if (number === 0) { 
                words = "Zero"; 
            } else { 
                if (Math.floor(number / 100) > 0) { 
                    words += singleDigits[Math.floor(number / 100)] + " Hundred "; 
                } 
                const remainder = number % 100; 

 
                if (remainder >= 10 && remainder < 20) { 
                    words += teens[remainder - 10]; 
                } else { 
                    if (Math.floor(remainder / 10) > 0) { 
                        words += tens[Math.floor(remainder / 10)] + " "; 
                    } 
                    if (remainder % 10 > 0) { 
                        words += singleDigits[remainder % 10]; 
                    } 
                } 
            } 
            outputDiv.textContent = "In words: " + words.trim(); 
        } 
    </script>
