    <p>Enter numbers separated by commas:</p> 
    <input type="text" id="numbers"> 
    <button onclick="calculateAverage()">Calculate Average</button> 

 
    <p id="avgResult"></p> 
    <script> 
        function calculateAverage() { 
            const input = document.getElementById("numbers").value; 
            const nums = input.split(',').map(Number).filter(n => !isNaN(n)); 
            if (nums.length === 0) { 
                document.getElementById("avgResult").innerText = "Please enter valid numbers."; 
                return; } 
            const sum = nums.reduce((acc, val) => acc + val, 0); 
            const avg = sum / nums.length; 
            document.getElementById("avgResult").innerText = "Average: " + avg.toFixed(2); 
        } 
    </script> 
</body>
