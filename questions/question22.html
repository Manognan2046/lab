Question 22

Write a program that uses a 2D array to store a matrix and calculates the sum of each

Code / Solution from the Record

<!DOCTYPE html> 
<html lang="en"> 
<head> 
    <title>Matrix Column Sum</title> 

 
</head> 
<body style="text-align: center;"> 
    <h2>Matrix Column Sum Calculator</h2> 
    <p>Enter matrix rows (comma-separated numbers, one row per line):</p> 
    <textarea id="matrixInput" rows="5" cols="40"></textarea><br><br> 
    <button onclick="calculateColumnSums()">Calculate Column Sums</button> 
    <p id="matrixResult"></p> 
 
    <script> 
        function calculateColumnSums() { 
            const rawInput = document.getElementById("matrixInput").value.trim(); 
            const rows = rawInput.split('\n').map(row => row.split(',').map(Number)); 
 
            if (rows.length === 0 || rows.some(r => r.some(isNaN))) { 
                document.getElementById("matrixResult").innerText = "Invalid matrix input."; 
                return; 
            } 
            const cols = rows[0].length; 
            const sums = Array(cols).fill(0); 
 
            for (let r of rows) { 
                if (r.length !== cols) { 
                    document.getElementById("matrixResult").innerText = "Rows must have same 
number of columns."; 
                    return; 
                } 
                for (let i = 0; i < cols; i++) { 

 
                    sums[i] += r[i]; 
                } 
            } 
            document.getElementById("matrixResult").innerText = "Column Sums: " + 
sums.join(", "); 
        } 
    </script> 
</body> 
</html>
