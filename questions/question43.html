using System; 
 
namespace ProductCalculator 
{ 
    class Program 
    { 
        static void Main(string[] args) 
        { 

 
            int[] numbers = new int[4]; 
            int product = 1; 
 
            Console.WriteLine("Enter four integer values:"); 
 
            for (int i = 0; i < 4; i++) 
            { 
                Console.Write("Enter number {0}: ", i + 1); 
                string input = Console.ReadLine(); 
 
                // Input validation 
                while (!int.TryParse(input, out numbers[i])) 
                { 
                    Console.Write("Invalid input. Enter an integer for number {0}: ", i + 1); 
                    input = Console.ReadLine(); 
                } 
 
                product *= numbers[i]; 
            } 
 
            Console.WriteLine("The product of the entered numbers is: " + product); 
        } 
    } 
}
